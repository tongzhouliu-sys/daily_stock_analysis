import { test, expect } from "@playwright/test";
import fs from "node:fs";
import path from "node:path";

test("read-only screen charts, twelve rules, motion control and small screens", async ({
  page,
}) => {
  const errors: string[] = [];
  page.on("pageerror", (e) => errors.push(e.message));
  await page.setViewportSize({ width: 1920, height: 1080 });
  await page.goto("/screen");
  await page
    .getByLabel("本机访问密码")
    .fill(
      fs
        .readFileSync(
          path.resolve(
            process.env.SCANNER_BROWSER_TEST_ROOT || "../runtime/browser-test",
            "首次登录密码.txt",
          ),
          "utf8",
        )
        .trim(),
    );
  await page.getByRole("button", { name: "进入工作台" }).click();
  await expect(
    page.getByRole("heading", { name: "市场信号全景", exact: true }),
  ).toBeVisible();
  await expect(page.locator(".sc-rule")).toHaveCount(12);
  await expect(page.locator(".sc-source")).toHaveCount(3);
  await expect(page.locator(".sc-chart-area")).toBeVisible();
  await page.getByRole("button", { name: "3M", exact: true }).click();
  await expect(
    page.getByRole("button", { name: "3M", exact: true }),
  ).toHaveClass("active");
  await page.getByRole("button", { name: "关闭动态效果" }).click();
  await expect(page.locator(".screen-shell")).not.toHaveClass(/is-motion/);
  await page.screenshot({
    path: "../runtime/e2e-results/screen-desktop.png",
    fullPage: true,
  });
  for (const width of [768, 375]) {
    await page.setViewportSize({ width, height: 950 });
    expect(
      await page.evaluate(
        () => document.documentElement.scrollWidth <= innerWidth,
      ),
    ).toBe(true);
    await page.screenshot({
      path: `../runtime/e2e-results/screen-${width}.png`,
      fullPage: true,
    });
  }
  await page.getByRole("link", { name: "返回工作台", exact: true }).click();
  await expect(
    page.getByRole("heading", { name: "今日工作台", exact: true }),
  ).toBeVisible();
  expect(errors).toEqual([]);
});

import { test, expect } from "@playwright/test";
import fs from "node:fs";
import path from "node:path";

test("five real pages, decision logging, separate paper account and responsive risk visibility", async ({
  page,
}) => {
  const errors: string[] = [];
  page.on("pageerror", (e) => errors.push(e.message));
  await page.goto("/");
  await page
    .getByLabel("本机访问密码")
    .fill(
      fs
        .readFileSync(
          path.resolve(
            process.env.SCANNER_BROWSER_TEST_ROOT || "../runtime/browser-test",
            "首次登录密码.txt",
          ),
          "utf8",
        )
        .trim(),
    );
  await page.getByRole("button", { name: "进入工作台" }).click();
  await expect(
    page.getByRole("heading", { name: "今日工作台", exact: true }),
  ).toBeVisible();
  await expect(page.getByText("2 / 2", { exact: true })).toBeVisible();
  await expect(
    page.getByText("当前是虚构验证环境", { exact: false }),
  ).toBeVisible();
  await page.screenshot({
    path: "../runtime/e2e-results/workspace-desktop.png",
    fullPage: true,
  });
  await page.getByRole("link", { name: "信号与观察", exact: true }).click();
  await expect(
    page.getByRole("heading", { name: "信号与观察", exact: true }),
  ).toBeVisible();
  await page.getByRole("link", { name: "DEMO", exact: true }).first().click();
  await expect(
    page.getByRole("heading", { name: "记录你的判断", exact: true }),
  ).toBeVisible();
  await page
    .getByLabel("理由", { exact: true })
    .fill("虚构验证：仅等待结构，不产生订单");
  await page.getByLabel("预期", { exact: true }).fill("后续按完成会话核对");
  await page.getByLabel("证伪条件", { exact: true }).fill("跌回冻结关键位置");
  await page.getByRole("button", { name: "等待", exact: true }).click();
  await expect(
    page.getByText("判断已记录；没有产生订单或预算占用。"),
  ).toBeVisible();
  await page.screenshot({
    path: "../runtime/e2e-results/signal-detail.png",
    fullPage: true,
  });
  await page.getByRole("link", { name: "设置与数据", exact: true }).click();
  await page.getByLabel("名称", { exact: true }).fill("浏览器验证纸面账户");
  await page.getByLabel("明确的模拟初始资金（USD）").fill("50000");
  await page.getByRole("button", { name: "建立账户", exact: true }).click();
  await expect(
    page.getByText(
      "账户已建立。请查看并确认当前账户快照完整；纸面账户不是真实资产。",
    ),
  ).toBeVisible();
  await page.getByRole("link", { name: "日志与复盘", exact: true }).click();
  await expect(
    page.getByText("虚构验证：仅等待结构，不产生订单").first(),
  ).toBeVisible();
  await page.screenshot({
    path: "../runtime/e2e-results/journal-desktop.png",
    fullPage: true,
  });
  await page.setViewportSize({ width: 375, height: 850 });
  await page.getByRole("link", { name: "今日工作台", exact: true }).click();
  await expect(
    page.getByRole("heading", { name: "隔夜风险与账户核对" }),
  ).toBeVisible();
  expect(
    await page.evaluate(
      () => document.documentElement.scrollWidth <= window.innerWidth,
    ),
  ).toBe(true);
  await page.screenshot({
    path: "../runtime/e2e-results/workspace-mobile.png",
    fullPage: true,
  });
  await page.getByRole("link", { name: "信号与观察", exact: true }).click();
  expect(
    await page.evaluate(
      () => document.documentElement.scrollWidth <= window.innerWidth,
    ),
  ).toBe(true);
  await page.screenshot({
    path: "../runtime/e2e-results/signals-mobile.png",
    fullPage: true,
  });
  await page.setViewportSize({ width: 768, height: 1024 });
  expect(
    await page.evaluate(
      () => document.documentElement.scrollWidth <= window.innerWidth,
    ),
  ).toBe(true);
  await page.screenshot({
    path: "../runtime/e2e-results/signals-tablet.png",
    fullPage: true,
  });
  await page.keyboard.press("Tab");
  expect(await page.evaluate(() => document.activeElement?.tagName)).not.toBe(
    "BODY",
  );
  expect(errors).toEqual([]);
});
